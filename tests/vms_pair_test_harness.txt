#!/bin/bash
# /home/claude/vmstest/run_pair.sh  v6 (old DMC debug files removed before each run)
# v5: VMS names the DMR11 line DMC-0, like the DMC11
# v4: circuit check: the other node is reached over the circuit
# v3: circuit check via node reachability; DIFFERENCES result line
# v2: pkill by process name (-f also matched this script)
# Two-node DECnet test for the DMC/DMR/DMP11 and DMF32 emulations on two
# VAX-11/780 simulators, with fresh copies of Ulli's VMS disks.
#
#   run_pair.sh VMS DEV SIM [DEV2]
#     VMS  47 (V4.7, RA81, nodes MV1/MV2) or 37 (V3.7, RK07, nodes VAX1/VAX2)
#     DEV  DMC | DMR | DMP | DMF   device of node 1 (and node 2)
#     SIM  simulator binary (vax780.v6 = before, vax780.v7 = with the DMC fixes)
#     DEV2 optional other device for node 2 (mixed pair, e.g. DMF with DMC)
#
# Steps: boot both nodes, log in as SYSTEM, replace the permanent DECnet
# line/circuit with one point-to-point circuit on DEV, STARTNET, wait for
# the circuit, then:
#   1. both nodes copy HELPLIB.HLB to the other node at the same time
#   2. node 1 fetches its copy back and compares it (DIFFERENCES)
#   3. node 2 does the same
#   4. node 1 turns its circuit off and on; the circuit must come back and
#      a second copy + DIFFERENCES must pass
#   5. circuit counters of both nodes
# Results go to stdout; the consoles are in n1/pane.log and n2/pane.log.
VMS=$1; DEV=$2; SIM=$3; DEV2=${4:-$2}
T=/home/claude/vmstest
V="python3 -I $T/vcon.py"
cd $T

# ---- device-specific names --------------------------------------------
line_of() {     # VMS DECnet line name for a device
  case $1 in DMC) echo DMC-0;; DMR) echo DMC-0;;   # XMDRIVER: DMC11 and DMR11 are both DMC-n
                  DMP) echo DMP-0;; DMF) echo DMF-0;; esac
}
if [ $VMS = 47 ]; then NAME1=MV1; NAME2=MV2; DISK=R81; HLB=2235; else NAME1=VAX1; NAME2=VAX2; DISK=RK07; HLB=1734; fi

# ---- stop old runs, wait for free ports ----------------------------------
tmux kill-server 2>/dev/null; pkill -9 -x vax780.v6; pkill -9 -x vax780.v7; sleep 1
python3 - <<'EOF'
import socket, time
while True:
    busy = False
    for p in (1111, 2222):
        s = socket.socket(); s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
        try: s.bind(('0.0.0.0', p))
        except OSError: busy = True
        s.close()
    if not busy: break
    time.sleep(2)
EOF

# ---- fresh disks and ini files -------------------------------------------
for n in 1 2; do
  if [ $n = 1 ]; then D=$DEV; MY=1111; PEER=2222; else D=$DEV2; MY=2222; PEER=1111; fi
  rm -f n$n/*.R81 n$n/*.RK07 n$n/node$n.log n$n/dmc$n.dbg
  if [ $VMS = 47 ]; then cp disks/VMS047-$n.R81 n$n/; else cp disks/VMS037-$n.RK07 n$n/; fi
  chmod u+w n$n/VMS0*
  {
    echo "; /home/claude/vmstest/n$n/pair.ini  (generated by run_pair.sh v1: VMS V$VMS, $D)"
    echo "SET CONSOLE LOG=node$n.log"
    echo "SET CPU IDLE=VMS"
    if [ $VMS = 47 ]; then
      echo "SET RQ ENABLE"; echo "SET RQ0 RA81"; echo "ATT RQ0 VMS047-$n.R81"
    else
      echo "SET HK ENABLE"; echo "SET HK0 RK07"; echo "ATT HK0 VMS037-$n.RK07"
    fi
    case $D in
      DMC|DMR|DMP)
        echo "SET DMC ENABLE"; echo "SET DMC0 TYPE=$D"
        echo "SET DMC0 PEER=LOCALHOST:$PEER"; echo "ATTACH DMC0 $MY"
        echo "SET DMC DEBUG=INFO;WARN"; echo "SET DEBUG -N dmc$n.dbg";;
      DMF)
        echo "SET DMF ENABLE"; echo "SET DMF PEER=LOCALHOST:$PEER"; echo "ATTACH DMF3 $MY";;
    esac
    if [ $VMS = 47 ]; then echo "BOOT RQ0"; else echo "BOOT HK0"; fi
  } > n$n/pair.ini
  $V $n start pair.ini $T/$SIM
done
echo "=== VMS V$VMS, node 1 $DEV ($(line_of $DEV)), node 2 $DEV2 ($(line_of $DEV2)), simulator $SIM"

# ---- boot, log in, configure DECnet, start it ----------------------------
for n in 1 2; do
  if [ $n = 1 ]; then L=$(line_of $DEV); else L=$(line_of $DEV2); fi
  $V $n wait "job terminated" 400 > /dev/null || { echo "node $n: no boot"; exit 1; }
  sleep 2
  $V $n cmd "" "Username: *$" 30 > /dev/null
  $V $n cmd "SYSTEM" '\$ $' 60 > /dev/null
  for c in "MC NCP PURGE KNOWN CIRCUITS ALL" "MC NCP PURGE KNOWN LINES ALL" \
           "MC NCP DEFINE LINE $L STATE ON PROTOCOL DDCMP POINT" \
           "MC NCP DEFINE CIRCUIT $L STATE ON" "MC NCP DEFINE EXECUTOR STATE ON" \
           "SHOW DEVICE X"; do
    $V $n cmd "$c" | tr -d '\000' | grep -a "%\|^X\|_X" | grep -av "^\\$"
  done
done
for n in 1 2; do $V $n cmd '@SYS$MANAGER:STARTNET' '\$ $' 300 | tr -d '\000' | grep -a "%" ; done

# ---- wait until the circuit is up on node 1 ------------------------------
L1=$(line_of $DEV); L2=$(line_of $DEV2)
wait_up() {     # $1 = node, $2 = circuit (for the message), $3 = adjacent node name
  # V4.7 does not list the adjacent node in the SHOW CIRCUIT summary on
  # every node, and an end node shows no "reachable" state, so check that
  # SHOW NODE names this circuit for the other node.
  for i in $(seq 1 40); do
    if $V $1 cmd "MC NCP SHOW NODE $3" | tr -d '\000' | grep -aq "($3) .*$2"; then echo up; return 0; fi
    sleep 5
  done
  echo DOWN; $V $1 cmd "MC NCP SHOW CIRCUIT $2" | tr -d '\000' | grep -a "  $2"; return 1
}
t0=$(date +%s)
echo "circuit $L1 on node 1: $(wait_up 1 $L1 "$NAME2") after $(( $(date +%s) - t0 )) s"
echo "circuit $L2 on node 2: $(wait_up 2 $L2 "$NAME1")"

# ---- 1-3: copies both ways at the same time, fetch back, compare -----------
diffs() {       # $1 = node: compare the file fetched back with the original
  $V $1 cmd 'DIFFERENCES SYS$HELP:HELPLIB.HLB SYS$LOGIN:BACK.HLB' '\$ $' 600 | tr -d '\000' |
    grep -a "sections found\|%" | head -2
}
t0=$(date +%s)
$V 1 mark; $V 2 mark
$V 1 raw 'COPY SYS$HELP:HELPLIB.HLB '"$NAME2"'"SYSTEM"::SYS$LOGIN:FROM1.HLB'; tmux send-keys -t v1 C-m
$V 2 raw 'COPY SYS$HELP:HELPLIB.HLB '"$NAME1"'"SYSTEM"::SYS$LOGIN:FROM2.HLB'; tmux send-keys -t v2 C-m
r1=$($V 1 wait '\$ $' 900 | tr -d '\000' | grep -a "%" | head -2)
r2=$($V 2 wait '\$ $' 900 | tr -d '\000' | grep -a "%" | head -2)
echo "1. simultaneous copies of HELPLIB.HLB ($HLB blocks) each way: $(( $(date +%s) - t0 )) s ${r1:+node1: $r1} ${r2:+node2: $r2}"
$V 1 cmd 'COPY '"$NAME2"'"SYSTEM"::SYS$LOGIN:FROM1.HLB SYS$LOGIN:BACK.HLB' '\$ $' 900 | tr -d '\000' | grep -a "%"
echo "2. node 1 fetched its copy back: $(diffs 1)"
$V 2 cmd 'COPY '"$NAME1"'"SYSTEM"::SYS$LOGIN:FROM2.HLB SYS$LOGIN:BACK.HLB' '\$ $' 900 | tr -d '\000' | grep -a "%"
echo "3. node 2 fetched its copy back: $(diffs 2)"

# ---- 4: circuit off/on ------------------------------------------------------
$V 1 cmd "MC NCP SET CIRCUIT $L1 STATE OFF" | tr -d '\000' | grep -a "%"
sleep 10
$V 1 cmd "MC NCP SET CIRCUIT $L1 STATE ON" | tr -d '\000' | grep -a "%"
t0=$(date +%s)
echo "4. circuit off/on on node 1: $(wait_up 1 $L1 "$NAME2") after $(( $(date +%s) - t0 )) s"
$V 1 cmd 'DELETE SYS$LOGIN:BACK.HLB;*' > /dev/null
$V 1 cmd 'COPY SYS$HELP:HELPLIB.HLB '"$NAME2"'"SYSTEM"::SYS$LOGIN:AGAIN.HLB' '\$ $' 900 | tr -d '\000' | grep -a "%"
$V 1 cmd 'COPY '"$NAME2"'"SYSTEM"::SYS$LOGIN:AGAIN.HLB SYS$LOGIN:BACK.HLB' '\$ $' 900 | tr -d '\000' | grep -a "%"
echo "   copy after the restart: $(diffs 1)"

# ---- 5: counters ----------------------------------------------------------
for n in 1 2; do
  if [ $n = 1 ]; then L=$L1; else L=$L2; fi
  echo "5. node $n circuit $L counters:"
  $V $n cmd "MC NCP SHOW CIRCUIT $L COUNTERS" | tr -d '\000' |
    grep -aiE "error|timeout|buffer unavail|blocks|messages|Circuit =|down|failure|rejected" | sed 's/^/     /'
done
for n in 1 2; do [ -f n$n/dmc$n.dbg ] && echo "   node $n master clears: $(grep -ac 'Master clear' n$n/dmc$n.dbg)"; done
echo "=== end"

#!/bin/bash
# /home/claude/vmstest/run_all.sh  v2 (restarted at the DMR tests) - all VMS V4.7 / V3.7 pair tests in sequence.
# One result file per test: r<VMS>_<DEV>[_<DEV2>]_<build>.txt
cd /home/claude/vmstest
while read -r vms dev sim dev2; do
  [ -z "$vms" ] && continue
  out=r${vms}_${dev}${dev2:+_$dev2}_${sim#vax780.}.txt
  ./run_pair.sh $vms $dev $sim $dev2 > $out 2>&1
  for n in 1 2; do cp n$n/pane.log ${out%.txt}_n$n.log; done
done <<'LIST'
47 DMR vax780.v7
47 DMR vax780.v6
47 DMP vax780.v7
47 DMF vax780.v7
47 DMF vax780.v7 DMC
37 DMC vax780.v7
37 DMC vax780.v6
37 DMR vax780.v7
37 DMR vax780.v6
37 DMP vax780.v7
LIST
echo ALLDONE > run_all.done

# /home/claude/vmstest/vcon.py  v2 (VMS prompt arrives as CR NUL "$ ")
# Console driver for several SIMH nodes, each running in tmux session v<N>
# with its console copied (tmux pipe-pane) to /home/claude/vmstest/n<N>/pane.log.
#
#   python3 vcon.py N start INI [SIM]   start node N (fresh pane.log) with INI
#   python3 vcon.py N cmd "text" [regex] [s]
#                                       mark the log, type text + CR, wait for
#                                       regex (default VMS prompt "$ "), print
#                                       the output since the mark
#   python3 vcon.py N raw "^E"          type a control character / text w/o CR
#   python3 vcon.py N wait regex [s]    wait for regex after the last mark
#   python3 vcon.py N mark              set the mark to the end of the log
#   python3 vcon.py N stop              ^E, EXIT the simulator, kill session
#
# Based on /home/claude/ult/run/con.py v3 (a trailing ";" is sent as a hex
# key because tmux takes it as a command separator).
import os, re, subprocess, sys, time

BASE = '/home/claude/vmstest'

def paths(n):
    d = os.path.join(BASE, 'n%s' % n)
    return d, os.path.join(d, 'pane.log'), os.path.join(d, 'pane.log.pos')

def tmux(n, *keys):
    subprocess.run(['tmux', 'send-keys', '-t', 'v%s' % n] + list(keys), check=True)

def read_log(n):
    try:
        with open(paths(n)[1], 'rb') as f:
            return f.read().decode('latin1').replace('\r', '')
    except FileNotFoundError:
        return ''

def mark(n):
    open(paths(n)[2], 'w').write(str(len(read_log(n))))

def get_mark(n):
    try:
        return int(open(paths(n)[2]).read())
    except Exception:
        return 0

def send(n, text, cr=True):
    m = re.fullmatch(r'\^([A-Za-z])', text)
    if m:
        tmux(n, 'C-' + m.group(1).lower())
        return
    tail = ''
    while text.endswith(';'):
        text, tail = text[:-1], tail + ';'
    if text:
        tmux(n, '-l', text)
    for _ in tail:
        tmux(n, '-H', '3b')
    if cr:
        tmux(n, 'C-m')

def wait(n, pattern, timeout):
    start = get_mark(n); t0 = time.time(); rx = re.compile(pattern)
    while True:
        data = read_log(n)
        m = rx.search(data, start)
        if m:
            sys.stdout.write(data[start:m.end()] + '\n')
            open(paths(n)[2], 'w').write(str(m.end()))
            return 0
        if time.time() - t0 > timeout:
            sys.stdout.write(data[start:] + '\n*** TIMEOUT (node %s) waiting for %r ***\n' % (n, pattern))
            return 1
        time.sleep(0.5)

if __name__ == '__main__':
    n, op = sys.argv[1], sys.argv[2]
    d, log, pos = paths(n)
    if op == 'start':
        ini = sys.argv[3]
        sim = sys.argv[4] if len(sys.argv) > 4 else BASE + '/vax780.v7'
        subprocess.run(['tmux', 'kill-session', '-t', 'v%s' % n], stderr=subprocess.DEVNULL)
        for p in (log, pos):
            if os.path.exists(p):
                os.remove(p)
        subprocess.run(['tmux', 'new-session', '-d', '-s', 'v%s' % n, '-x', '132', '-y', '50',
                        'cd %s && %s %s; sleep 999999' % (d, sim, ini)], check=True)
        subprocess.run(['tmux', 'pipe-pane', '-o', '-t', 'v%s' % n, 'cat >> %s' % log], check=True)
    elif op == 'cmd':
        mark(n)
        send(n, sys.argv[3])
        sys.exit(wait(n, sys.argv[4] if len(sys.argv) > 4 else r'\n\x00*\$ $',
                      int(sys.argv[5]) if len(sys.argv) > 5 else 120))
    elif op == 'raw':
        send(n, sys.argv[3], cr=False)
    elif op == 'wait':
        sys.exit(wait(n, sys.argv[3], int(sys.argv[4]) if len(sys.argv) > 4 else 300))
    elif op == 'mark':
        mark(n)
    elif op == 'stop':
        send(n, '^E'); time.sleep(1); send(n, 'EXIT'); time.sleep(2)
        subprocess.run(['tmux', 'kill-session', '-t', 'v%s' % n], stderr=subprocess.DEVNULL)
